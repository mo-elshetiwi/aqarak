## Supplementary speech source groups

I generate the two source-group files from the committed text-free clip scores using the committed bootstrap function. I preserve clip order and the same 2,000-resample, seed-20260928 defaults. I run the following from the repository root to reproduce these files:

```sh
pnpm --filter @aqarak/evaluation exec tsx --input-type=module <<'JS'
import { readFileSync, writeFileSync } from "node:fs";
import { clusterBootstrapRatio, renderCsv, SCORER_VERSION } from "./src/scoring/index.ts";
const root = "results/held-out-2026-09-28/mc3_speech_to_text";
const data = JSON.parse(readFileSync(`${root}/scores.json`, "utf8"));
const clips = data.candidates.scores[0].clips;
const groups = [
  { name: "MIXAT", clips: clips.filter(c => !c.clipId.includes("/")) },
  { name: "FLEURS", clips: clips.filter(c => c.clipId.includes("/")) },
].map(g => ({
  source: g.name,
  items: g.clips.length,
  referenceWords: g.clips.reduce((s, c) => s + c.referenceWords, 0),
  wordEdits: g.clips.reduce((s, c) => s + c.wordEdits, 0),
  wer: clusterBootstrapRatio(g.clips.map(c => ({
    numerator: c.wordEdits, denominator: c.referenceWords,
  }))),
}));
writeFileSync(`${root}/source-summary.json`, JSON.stringify({
  scorerVersion: SCORER_VERSION, seed: 20260928, resamples: 2000,
  confidence: 0.95,
  method: "I apply the committed clusterBootstrapRatio to the text-free clip scores in frozen order.",
  groups,
}, null, 2) + "\n");
writeFileSync(`${root}/source-summary.csv`, renderCsv(
  ["source", "items", "reference_words", "word_edits", "wer", "wer_low", "wer_high"],
  groups.map(g => [g.source, g.items, g.referenceWords, g.wordEdits,
    g.wer?.estimate ?? null, g.wer?.lower ?? null, g.wer?.upper ?? null]),
));
JS
```
